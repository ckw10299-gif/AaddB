def aud_hash(p):
    return run(['ffmpeg','-v','error','-i',p,'-map','0:a:0','-c:a','copy','-hash','sha256','-f','hash','pipe:1'],capture_output=True).stdout

def verify(lang,size):

    old=OUT/'<PROJECT_VALUE>'/name
    new=STAGE/'<PROJECT_VALUE>'/name
    info,ref=probe(new),probe(old)
    v,a=info['streams']
    w,h=map(int,size.split('x'))
    assert (v['width'],v['height'],v['r_frame_rate'],v['codec_name'],a['codec_name'])==(w,h,'30/1','h264','aac')
    assert abs(float(v['duration'])-float(ref['streams'][0]['duration']))<PROJECT_NUMBER
    assert a['duration_ts']==ref['streams'][1]['duration_ts'] and aud_hash(new)==aud_hash(old)
    run(['ffmpeg','-v','error','-xerror','-threads',3,'-i',new,'-f','null','NUL'],capture_output=True)
    oldtail=frame(old,PROJECT_NUMBER,PROJECT_NUMBER,PROJECT_NUMBER); newtail=frame(new,PROJECT_NUMBER,PROJECT_NUMBER,PROJECT_NUMBER)
    assert max(ImageStat.Stat(ImageChops.difference(oldtail,newtail)).mean)<2

    t0,t1,_=max(cues_for(lang),key=lambda c:len(c[2]))
