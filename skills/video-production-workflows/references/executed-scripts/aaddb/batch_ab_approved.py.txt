def probe(path):
    raw = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration:stream=codec_type,width,height", "-of", "json", str(path)],
        check=True, capture_output=True, text=True, encoding="utf-8",
    ).stdout
    data = json.loads(raw)
    vs = next(s for s in data["streams"] if s["codec_type"] == "video")
    au = next((s for s in data["streams"] if s["codec_type"] == "audio"), None)
    return {"duration": float(data["format"]["duration"]), "width": int(vs["width"]), "height": int(vs["height"]), "audio": au is not None}

def zoom_expr(kind, dur, start_frame):
